cmake_minimum_required(VERSION 3.22.1)
project(VectrasBox VERSION 1.0.1 LANGUAGES C CXX)

add_compile_options(-O3 -ffp-contract=fast)

include(FetchContent)
foreach(dependency zlib libpng freetype libsndfile libslirp)
    string(TOUPPER "${dependency}" dependency_upper)
    get_filename_component(bundled_source "${CMAKE_CURRENT_LIST_DIR}/../../../../native-dependencies/${dependency}" ABSOLUTE)
    if(EXISTS "${bundled_source}/CMakeLists.txt" OR EXISTS "${bundled_source}/meson.build")
        set(FETCHCONTENT_SOURCE_DIR_${dependency_upper} "${bundled_source}" CACHE PATH "" FORCE)
    endif()
endforeach()
set(FETCHCONTENT_UPDATES_DISCONNECTED ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)

set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
add_subdirectory(SDL2 EXCLUDE_FROM_ALL)

set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
FetchContent_Declare(zlib GIT_REPOSITORY https://github.com/madler/zlib.git GIT_TAG v1.3.2 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(zlib)
if(TARGET zlibstatic AND NOT TARGET ZLIB::ZLIB)
    add_library(ZLIB::ZLIB ALIAS zlibstatic)
endif()

set(PNG_SHARED OFF CACHE BOOL "" FORCE)
set(PNG_STATIC ON CACHE BOOL "" FORCE)
set(PNG_TESTS OFF CACHE BOOL "" FORCE)
set(PNG_TOOLS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(libpng GIT_REPOSITORY https://github.com/pnggroup/libpng.git GIT_TAG v1.6.53 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(libpng)
if(TARGET png_static AND NOT TARGET PNG::PNG)
    add_library(PNG::PNG ALIAS png_static)
endif()

set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
FetchContent_Declare(freetype GIT_REPOSITORY https://gitlab.freedesktop.org/freetype/freetype.git GIT_TAG VER-2-13-3 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(freetype)

set(BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ENABLE_EXTERNAL_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_MPEG OFF CACHE BOOL "" FORCE)
set(ENABLE_CPACK OFF CACHE BOOL "" FORCE)
FetchContent_Declare(libsndfile GIT_REPOSITORY https://github.com/libsndfile/libsndfile.git GIT_TAG 1.2.2 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(libsndfile)
file(MAKE_DIRECTORY "${libsndfile_BINARY_DIR}/include")

if(NOT TARGET PkgConfig::FREETYPE)
    add_library(PkgConfig::FREETYPE INTERFACE IMPORTED GLOBAL)
    set_property(TARGET PkgConfig::FREETYPE PROPERTY INTERFACE_LINK_LIBRARIES freetype)
endif()
if(NOT TARGET PkgConfig::SNDFILE)
    add_library(PkgConfig::SNDFILE INTERFACE IMPORTED GLOBAL)
    set_property(TARGET PkgConfig::SNDFILE PROPERTY INTERFACE_LINK_LIBRARIES sndfile)
    set_property(TARGET PkgConfig::SNDFILE PROPERTY INTERFACE_INCLUDE_DIRECTORIES "${libsndfile_SOURCE_DIR}/include")
endif()

set(QT OFF CACHE BOOL "" FORCE)
set(SDL2 ON CACHE BOOL "" FORCE)
set(RELEASE ON CACHE BOOL "" FORCE)
set(OPENAL OFF CACHE BOOL "" FORCE)
set(RTMIDI OFF CACHE BOOL "" FORCE)
set(FLUIDSYNTH OFF CACHE BOOL "" FORCE)
set(MUNT ON CACHE BOOL "" FORCE)
set(SOUNDCANVAS OFF CACHE BOOL "" FORCE)
set(DISCORD OFF CACHE BOOL "" FORCE)
set(VNC OFF CACHE BOOL "" FORCE)
set(CPPTHREADS ON CACHE BOOL "" FORCE)
set(USE_SDL_SHADER ON CACHE BOOL "" FORCE)
set(USE_IMGUI ON CACHE BOOL "" FORCE)

include(cmake/PCBoxAndroid.cmake)
include(cmake/LibSlirpAndroid.cmake)
vectras_add_pcbox()
option(VECTRAS_BUILD_BACKEND_TESTS "Build standalone Android backend checks" OFF)
if(VECTRAS_BUILD_BACKEND_TESTS)
    include(cmake/AndroidBackendTests.cmake)
endif()
